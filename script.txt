// ===============================
// DEFAULT TODAY DATE
// ===============================

const today = new Date();

const yyyy =
  today.getFullYear();

const mm =
  String(today.getMonth() + 1)
  .padStart(2, '0');

const dd =
  String(today.getDate())
  .padStart(2, '0');

const formattedDate =
  `${yyyy}-${mm}-${dd}`;





// ===============================
// API URL
// ===============================

const API_URL =
"https://script.google.com/macros/s/AKfycbxnEsuukKXHEIr4ym-3gryoFOkEXxlsR9a4VeMsxz1YJldj5NIJXvCTEiCwkqETcuT-/exec";





// ===============================
// GLOBAL ARRAYS
// ===============================

let absent = [];

let hwIncomplete = [];





// ===============================
// MOBILE VIBRATION
// ===============================

function vibrate(){

  if(navigator.vibrate){

    navigator.vibrate(40);
  }
}





// ===============================
// LOAD STUDENTS
// ===============================

async function loadStudents(){


  // SET TODAY DATE
  document
  .getElementById("attendanceDate")
  .value = formattedDate;



  try{

    const response =
      await fetch(
        API_URL + "?action=students"
      );



    const students =
      await response.json();



    const container =
      document.getElementById("students");



    let html = "";



    students.forEach(student => {

      html += `

        <div class="student-card">

          <div class="student-name">
            ${student.name}
          </div>

          <div class="row">

            <div
              class="box present"
              id="att-${student.id}">

              ✅ Present

            </div>



            <div
              class="box hwdone"
              id="hw-${student.id}">

              📘 HW Done

            </div>

          </div>

        </div>

      `;
    });



    container.innerHTML = html;





    // ===========================
    // ATTENDANCE EVENTS
    // ===========================

    students.forEach(student => {

      const attBox =
        document.getElementById(`att-${student.id}`);



      attBox.addEventListener("click", () => {

        toggleAttendance(student.id);

      });





      // ===========================
      // HW EVENTS
      // ===========================

      const hwBox =
        document.getElementById(`hw-${student.id}`);



      hwBox.addEventListener("click", () => {

        toggleHW(student.id);

      });

    });

  }

  catch(error){

    console.error(error);

    alert("Failed to load students");
  }
}





// ===============================
// TOGGLE ATTENDANCE
// ===============================

function toggleAttendance(id){

  vibrate();



  const attBox =
    document.getElementById(`att-${id}`);



  const hwBox =
    document.getElementById(`hw-${id}`);





  // ===========================
  // ALREADY ABSENT
  // ===========================

  if(absent.includes(id)){



    absent =
      absent.filter(x => x != id);



    attBox.className =
      "box present";



    attBox.innerHTML =
      "✅ Present";





    // RESET HW
    hwIncomplete =
      hwIncomplete.filter(x => x != id);



    hwBox.className =
      "box hwdone";



    hwBox.innerHTML =
      "📘 HW Done";





    // ENABLE HW
    hwBox.style.opacity = "1";



    hwBox.style.pointerEvents =
      "auto";
  }





  // ===========================
  // MAKE ABSENT
  // ===========================

  else{

    absent.push(id);



    attBox.className =
      "box absent";



    attBox.innerHTML =
      "❌ Absent";





    // AUTO HW MISSING
    if(!hwIncomplete.includes(id)){

      hwIncomplete.push(id);
    }



    hwBox.className =
      "box hwmiss";



    hwBox.innerHTML =
      "⚠️ HW Missing";





    // DISABLE HW CLICK
    hwBox.style.opacity =
      "0.6";



    hwBox.style.pointerEvents =
      "none";
  }
}





// ===============================
// TOGGLE HW
// ===============================

function toggleHW(id){

  vibrate();



  const box =
    document.getElementById(`hw-${id}`);





  // ===========================
  // HW DONE
  // ===========================

  if(hwIncomplete.includes(id)){



    hwIncomplete =
      hwIncomplete.filter(x => x != id);



    box.className =
      "box hwdone";



    box.innerHTML =
      "📘 HW Done";
  }





  // ===========================
  // HW MISSING
  // ===========================

  else{

    hwIncomplete.push(id);



    box.className =
      "box hwmiss";



    box.innerHTML =
      "⚠️ HW Missing";
  }
}





// ===============================
// LOAD EXISTING DATE DATA
// ===============================

async function loadAttendanceData(){

  try{



    // RESET ARRAYS
    absent = [];

    hwIncomplete = [];




    const selectedDate =
      document
      .getElementById("attendanceDate")
      .value;





    const response =
      await fetch(

        API_URL +

        "?action=attendance&date=" +

        selectedDate
      );





    const data =
      await response.json();





    // ===========================
    // RESET ALL UI
    // ===========================

    document
    .querySelectorAll("[id^='att-']")
    .forEach(box => {

      box.className =
        "box present";



      box.innerHTML =
        "✅ Present";

    });





    document
    .querySelectorAll("[id^='hw-']")
    .forEach(box => {

      box.className =
        "box hwdone";



      box.innerHTML =
        "📘 HW Done";



      box.style.opacity =
        "1";



      box.style.pointerEvents =
        "auto";
    });





    // ===========================
    // APPLY SAVED DATA
    // ===========================

    data.forEach(student => {

      const id =
        student.id;



      const attBox =
        document.getElementById(`att-${id}`);



      const hwBox =
        document.getElementById(`hw-${id}`);





      // ABSENT
      if(student.attendance == "A"){



        absent.push(id);



        attBox.className =
          "box absent";



        attBox.innerHTML =
          "❌ Absent";



        hwBox.style.opacity =
          "0.6";



        hwBox.style.pointerEvents =
          "none";
      }





      // HW MISSING
      if(student.hw == "X"){



        hwIncomplete.push(id);



        hwBox.className =
          "box hwmiss";



        hwBox.innerHTML =
          "⚠️ HW Missing";
      }

    });

  }

  catch(error){

    console.error(error);

    alert("Failed to load attendance data");
  }
}





// ===============================
// SAVE DATA
// ===============================

async function saveData(){

  vibrate();



  const btn =
    document.getElementById("saveBtn");



  const status =
    document.getElementById("saveStatus");





  // SELECTED DATE
  const selectedDate =
    document
    .getElementById("attendanceDate")
    .value;





  // LOADING UI
  btn.innerHTML =
    "Saving...";



  btn.style.background =
    "#666";





  try{

    await fetch(API_URL, {

      method:"POST",

      body: JSON.stringify({

        date: selectedDate,

        absent: absent,

        hwIncomplete: hwIncomplete

      })

    });





    // SUCCESS UI
    btn.innerHTML =
      "✓ Saved";



    btn.style.background =
      "green";



    status.innerHTML =
      "Data saved successfully";





    setTimeout(() => {

      btn.innerHTML =
        "SAVE DATA";



      btn.style.background =
        "#111";



      status.innerHTML =
        "";

    }, 2000);

  }

  catch(error){

    console.error(error);





    // ERROR UI
    btn.innerHTML =
      "Save Failed";



    btn.style.background =
      "red";



    status.innerHTML =
      "Connection Error";
  }
}





// ===============================
// SAVE BUTTON EVENT
// ===============================

document
.getElementById("saveBtn")
.addEventListener("click", saveData);





// ===============================
// DATE CHANGE EVENT
// ===============================

document
.getElementById("attendanceDate")
.addEventListener("change", loadAttendanceData);





// ===============================
// INITIALIZE APP
// ===============================

async function initializeApp(){

  await loadStudents();

  await loadAttendanceData();
}



initializeApp();
